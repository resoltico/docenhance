cmake_minimum_required(VERSION 4.4)
project(DocEnhanceJson LANGUAGES NONE)
if(NOT DOCENHANCE_JSON_BOUNDED_DESTRUCTION OR JSON_MultipleHeaders OR JSON_BuildTests)
  message(FATAL_ERROR "The reviewed bounded JSON destruction configuration is required")
endif()
set(de_header "${DE_UPSTREAM_SOURCE}/single_include/nlohmann/json.hpp")
file(SHA256 "${de_header}" de_digest)
if(NOT de_digest STREQUAL DOCENHANCE_JSON_HEADER_SHA256)
  message(FATAL_ERROR "Review bounded JSON destruction against the locked header")
endif()
file(READ "${de_header}" de_json)
set(de_begin "            if (t == value_t::array || t == value_t::object)\n            {\n                // flatten the current json_value to a heap-allocated stack")
set(de_end "\n            switch (t)\n")
string(FIND "${de_json}" "${de_begin}" de_first)
if(de_first LESS 0)
  message(FATAL_ERROR "The reviewed JSON cleanup sequence is absent")
endif()
string(SUBSTRING "${de_json}" ${de_first} -1 de_tail)
string(FIND "${de_tail}" "${de_end}" de_offset)
if(de_offset LESS 0)
  message(FATAL_ERROR "The reviewed JSON cleanup sequence is absent")
endif()
math(EXPR de_last "${de_first} + ${de_offset}")
string(SUBSTRING "${de_json}" 0 ${de_first} de_prefix)
string(SUBSTRING "${de_json}" ${de_last} -1 de_suffix)
set(de_owned "${CMAKE_CURRENT_BINARY_DIR}/owned-source")
file(MAKE_DIRECTORY "${de_owned}")
file(COPY "${DE_UPSTREAM_SOURCE}/CMakeLists.txt" "${DE_UPSTREAM_SOURCE}/cmake"
  "${DE_UPSTREAM_SOURCE}/single_include" "${DE_UPSTREAM_SOURCE}/nlohmann_json.natvis"
  DESTINATION "${de_owned}")
file(CONFIGURE OUTPUT "${de_owned}/single_include/nlohmann/json.hpp"
  CONTENT "${de_prefix}            // DocEnhance admits DOM depth before parsing. Recursive bounded cleanup\n            // must not allocate from this noexcept destruction boundary.\n${de_suffix}"
  @ONLY NEWLINE_STYLE LF)
add_subdirectory("${de_owned}" upstream)
