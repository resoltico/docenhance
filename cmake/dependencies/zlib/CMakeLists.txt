cmake_minimum_required(VERSION 4.4)
project(DocEnhanceZlib LANGUAGES C)
if(NOT DOCENHANCE_ZLIB_STREAM_ONLY OR ZLIB_BUILD_SHARED OR ZLIB_BUILD_TESTING)
  message(FATAL_ERROR "The private zlib build requires stream-only static configuration")
endif()
file(SHA256 "${DE_UPSTREAM_SOURCE}/CMakeLists.txt" digest)
if(NOT digest STREQUAL DOCENHANCE_ZLIB_CMAKE_SHA256)
  message(FATAL_ERROR "Review stream-only zlib against the locked upstream recipe")
endif()
add_subdirectory("${DE_UPSTREAM_SOURCE}" upstream)
get_target_property(sources zlibstatic SOURCES)
foreach(name IN ITEMS gzclose.c gzlib.c gzread.c gzwrite.c)
  list(FIND sources "${name}" index)
  if(index LESS 0)
    message(FATAL_ERROR "The reviewed zlib gzip source declaration is absent: ${name}")
  endif()
  list(REMOVE_ITEM sources "${name}")
endforeach()
set_property(TARGET zlibstatic PROPERTY SOURCES "${sources}")
